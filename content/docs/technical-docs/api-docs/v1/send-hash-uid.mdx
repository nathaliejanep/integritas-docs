| Request Type            | Content-Type                     | Body Type | Keys   | Description        |
| ----------------------- | -------------------------------- | --------- | ------ | ------------------ |
| {REQUEST_TYPE_RAW_DATA} | <code>{CONTENT_TYPE_JSON}</code> | `JSON`    | `hash` | Input hash of data |

```typescript
// Using fetch API
const API_KEY = 'your-api-key';
const REQUEST_ID = 'your-choice';

async function sendHash(hash: string) {
  const response = await fetch(
    'https://integritas.technology/core/v1/timestamp/post',
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': API_KEY,
        'x-request-id': REQUEST_ID,
      },
      body: JSON.stringify({ hash }),
    }
  );

  const data = await response.json();
  console.log('Response:', data);
  return data;
}

// Usage
const hash = 'E630FCDE204036E93323AE99843BF0EC6D03CAEB51CBBEA11E03D6739F8E9C3F';
sendHash(hash);
```

  <Tab value='Python'>

```python
import requests
import json

API_KEY = 'your-api-key'
REQUEST_ID = 'your-choice'

def send_hash(hash_value):
    url = 'https://integritas.technology/core/v1/timestamp/post'

    headers = {
        'Content-Type': 'application/json',
        'x-api-key': API_KEY,
        'x-request-id': REQUEST_ID,
    }

    payload = {'hash': hash_value}
    response = requests.post(url, headers=headers, json=payload)

    data = response.json()
    print('Response:', data)
    return data

# Usage
hash_value = 'E630FCDE204036E93323AE99843BF0EC6D03CAEB51CBBEA11E03D6739F8E9C3F'
send_hash(hash_value)
```

## API Response

The response from the API contains a UID which you will need to retrieve proof of the blockchain transaction that is being done by the stamp server.

### Example Response

```json
{
  "service": "CORE_API",
  "apiVersion": 1,
  "requestId": "send-hash-uid",
  "status": "success",
  "statusCode": 200,
  "message": "Hash forwarded successfully",
  "timestamp": "2026-01-16T14:31:03.573Z",
  "data": {
    "status": true,
    "data": "0xE630FCDE204036E93323AE99843BF0EC6D03CAEB51CBBEA11E03D6739F8E9C3F",
    "uid": "0x3B18508ABF8B0120CFEC"
  }
}
```
