  PLACEHOLDER_API_KEY,
  PLACEHOLDER_REQUEST_ID,
  INTEGRITAS_URL,

  DESC_X_REQUEST_ID,

} from '@/lib/constants';

| Attribute       | Details                                                                                                                             |
| --------------- | ----------------------------------------------------------------------------------------------------------------------------------- |
| **Method**      | <code>{HTTP_METHOD_POST}</code>                                                                                                     |
| **Description** | This endpoint submits a hash or returns verification data when the hash already exists on-chain. Send the hash as JSON in the body. |
| **Endpoint**    | <code>{API_BASE_URL}{ENDPOINT_V2_TIMESTAMP_POST}</code>                                                                             |

---

## Request Headers

| Header           | Value                                 | Description                                                                                              |
| ---------------- | ------------------------------------- | -------------------------------------------------------------------------------------------------------- |
| **Content-Type** | <code>{CONTENT_TYPE_JSON}</code>      | {DESC_CONTENT_TYPE_JSON}                                                                                 |
| **x-api-key**    | <code>{PLACEHOLDER_API_KEY}</code>    | {DESC_X_API_KEY} <a href={INTEGRITAS_URL} target="_blank" rel="noopener noreferrer">{INTEGRITAS_URL}</a> |
| **x-request-id** | <code>{PLACEHOLDER_REQUEST_ID}</code> | {DESC_X_REQUEST_ID}                                                                                      |

---

## Request Body

| Request Type            | Content-Type                     | Body Type | Keys   | Description        |
| ----------------------- | -------------------------------- | --------- | ------ | ------------------ |
| {REQUEST_TYPE_RAW_DATA} | <code>{CONTENT_TYPE_JSON}</code> | `JSON`    | `hash` | Input hash of data |

### Example

```json
{
  "hash": "a1b2c3d4e5f67890abcdef1234567890abcdef1234567890abcdef1234567891"
}
```

```typescript
const API_KEY = 'your-api-key';
const REQUEST_ID = 'your-choice';

async function timestampPost(hash: string) {
  const response = await fetch(
    'https://integritas.minima.global/core/v2/timestamp/post',
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': API_KEY,
        'x-request-id': REQUEST_ID,
      },
      body: JSON.stringify({ hash }),
    },
  );

  const data = await response.json();
  console.log('Response:', data);
  return data;
}

timestampPost(
  'a1b2c3d4e5f67890abcdef1234567890abcdef1234567890abcdef1234567891',
);
```

  <Tab value='Python'>

```python
import requests

API_KEY = 'your-api-key'
REQUEST_ID = 'your-choice'

def timestamp_post(hash_value):
    url = 'https://integritas.minima.global/core/v2/timestamp/post'
    headers = {
        'Content-Type': 'application/json',
        'x-api-key': API_KEY,
        'x-request-id': REQUEST_ID,
    }
    payload = {'hash': hash_value}
    response = requests.post(url, headers=headers, json=payload)
    data = response.json()
    print('Response:', data)
    return data

timestamp_post('a1b2c3d4e5f67890abcdef1234567890abcdef1234567890abcdef1234567891')
```
