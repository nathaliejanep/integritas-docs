## Request Body

| Request Type            | Content-Type                     | Body Type | Keys   | Description         |
| ----------------------- | -------------------------------- | --------- | ------ | ------------------- |
| {REQUEST_TYPE_RAW_DATA} | <code>{CONTENT_TYPE_JSON}</code> | `JSON`    | `hash` | Input hash to check |

```json
{
```

```typescript
const API_KEY = 'your-api-key';
const REQUEST_ID = 'your-choice';

async function fileCheck(hash: string) {
  const response = await fetch(
    'https://integritas.minima.global/core/v2/file/check',
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': API_KEY,
        'x-request-id': REQUEST_ID,
      },
      body: JSON.stringify({ hash }),
    },
  );

  const data = await response.json();
  console.log('Response:', data);
  return data;
}

fileCheck('a1b2c3d4e5f67890abcdef1234567890abcdef1234567890abcdef1234567890');
```

  <Tab value='Python'>

```python
import requests

API_KEY = 'your-api-key'
REQUEST_ID = 'your-choice'

def file_check(hash_value):
    url = 'https://integritas.minima.global/core/v2/file/check'
    headers = {
        'Content-Type': 'application/json',
        'x-api-key': API_KEY,
        'x-request-id': REQUEST_ID,
    }
    payload = {'hash': hash_value}
    response = requests.post(url, headers=headers, json=payload)
    data = response.json()
    print('Response:', data)
    return data

file_check('a1b2c3d4e5f67890abcdef1234567890abcdef1234567890abcdef1234567890')
```
