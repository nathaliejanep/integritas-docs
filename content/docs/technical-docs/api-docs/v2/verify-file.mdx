import {
  API_BASE_URL,
  ENDPOINT_V2_VERIFY_FILE,
  HTTP_METHOD_POST,
  CONTENT_TYPE_JSON,
  PLACEHOLDER_API_KEY,
  PLACEHOLDER_REQUEST_ID,
  INTEGRITAS_URL,
  DESC_X_API_KEY,
  DESC_X_REQUEST_ID,
  DESC_X_REPORT_REQUIRED,
  DESC_CONTENT_TYPE_JSON,
  REQUEST_TYPE_RAW_DATA,
} from '@/lib/constants';

| Attribute       | Details                                                                                                                           |
| --------------- | --------------------------------------------------------------------------------------------------------------------------------- |
| **Method**      | <code>{HTTP_METHOD_POST}</code>                                                                                                   |
| **Description** | This endpoint verifies a file by hash against the blockchain. Send the hash as JSON in the body. Optionally request a PDF report. |
| **Endpoint**    | <code>{API_BASE_URL}{ENDPOINT_V2_VERIFY_FILE}</code>                                                                              |

---

## Request Headers

| Header                | Value                                 | Description                                                                                              |
| --------------------- | ------------------------------------- | -------------------------------------------------------------------------------------------------------- |
| **Content-Type**      | <code>{CONTENT_TYPE_JSON}</code>      | {DESC_CONTENT_TYPE_JSON}                                                                                 |
| **x-api-key**         | <code>{PLACEHOLDER_API_KEY}</code>    | {DESC_X_API_KEY} <a href={INTEGRITAS_URL} target="_blank" rel="noopener noreferrer">{INTEGRITAS_URL}</a> |
| **x-report-required** | `true` / `false`                      | {DESC_X_REPORT_REQUIRED}                                                                                 |
| **x-request-id**      | <code>{PLACEHOLDER_REQUEST_ID}</code> | {DESC_X_REQUEST_ID}                                                                                      |

---

## Request Body

| Request Type            | Content-Type                     | Body Type | Keys       | Description                 |
| ----------------------- | -------------------------------- | --------- | ---------- | --------------------------- |
| {REQUEST_TYPE_RAW_DATA} | <code>{CONTENT_TYPE_JSON}</code> | `JSON`    | `hash`     | Input hash of file          |
| {REQUEST_TYPE_RAW_DATA} | <code>{CONTENT_TYPE_JSON}</code> | `JSON`    | `filename` | Optional. File name         |
| {REQUEST_TYPE_RAW_DATA} | <code>{CONTENT_TYPE_JSON}</code> | `JSON`    | `filesize` | Optional. File size (bytes) |

When provided, `filename` and `filesize` are used in the PDF report and in matched segments (the full file will be treated as one segment).

### Example

```json
{
  "hash": "106F1589B5818A400679A87A19214C43605DEC3EBFDD61E45A776A81C120DD9B",
  "filename": "test",
  "filesize": 5
}
```

  <Tab value='TypeScript'>

```typescript
const API_KEY = 'your-api-key';
const REQUEST_ID = 'your-choice';

async function verifyFile(hash: string, reportRequired = true) {
  const response = await fetch(
    'https://integritas.minima.global/core/v2/verify/file',
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': API_KEY,
        'x-report-required': reportRequired.toString(),
        'x-request-id': REQUEST_ID,
      },
      body: JSON.stringify({ hash }),
    },
  );

  const data = await response.json();
  console.log('Response:', data);
  return data;
}

verifyFile('106F1589B5818A400679A87A19214C43605DEC3EBFDD61E45A776A81C120DD9B');
```

  <Tab value='Python'>

```python
import requests

API_KEY = 'your-api-key'
REQUEST_ID = 'your-choice'

def verify_file(hash_value, report_required=True):
    url = 'https://integritas.minima.global/core/v2/verify/file'
    headers = {
        'Content-Type': 'application/json',
        'x-api-key': API_KEY,
        'x-report-required': str(report_required).lower(),
        'x-request-id': REQUEST_ID,
    }
    payload = {'hash': hash_value}
    response = requests.post(url, headers=headers, json=payload)
    data = response.json()
    print('Response:', data)
    return data

verify_file('106F1589B5818A400679A87A19214C43605DEC3EBFDD61E45A776A81C120DD9B')
```
